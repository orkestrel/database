import type { SQLiteValue } from '../sqlite/index.js'

// The SQLite database module — the server-native `DriverInterface` over the
// `src/server/sqlite` wrapper (never raw `node:sqlite`). The driver implements
// the core `DriverInterface`, so it adds no new public interface; the one type
// here is the `Criteria` → SQL compiler's output, the contract between
// `compilers.ts` and the driver. Types are the source of truth (AGENTS §2).

/**
 * A compiled SQL clause — the output of the `Criteria` → SQL compiler.
 *
 * @remarks
 * `sql` is the fragment that FOLLOWS the table name in a `SELECT * FROM <table>`
 * (a `WHERE` / `ORDER BY` / `LIMIT` tail, possibly empty); `params` are the bound
 * {@link SQLiteValue}s in clause order, one per `?` placeholder. Keeping the two
 * together lets the driver run `prepare(base + sql).all(params)` with no further
 * assembly — the values are already encoded for storage.
 */
export interface CompiledSQL {
	readonly sql: string
	readonly params: readonly SQLiteValue[]
}
