import { createDatabase, createRelationManager } from '@src/core'
import { createIndexedDBDriver } from '@src/browser'
import { afterEach, describe, expect, it } from 'vitest'
import {
	createUserRow,
	INTEGRATION_RELATIONS,
	INTEGRATION_TABLES,
	userRows,
} from '../../../setup.js'
import { createCleanups, createIntegrationDatabase, deleteDatabase } from '../../../setupBrowser.js'

// The whole core database + relations stack running over the IndexedDB driver in
// real Chromium — the proof that the driver is a true drop-in for the in-memory
// one: the Database / Table / Query / relations API is unchanged, only persistence
// differs. The cross-driver opener (`createIntegrationDatabase`, binding
// `createIndexedDBDriver` to `createDatabase` over the shared `INTEGRATION_TABLES`
// — tests/setup.ts) lives in `setupBrowser.ts` (§16.1), shared with the SQLite
// integration test so the two prove driver-parity over identical tables.

const cleanups = createCleanups()

// Open the cross-driver fixture and register its cleanup with this file's teardown.
function database() {
	const { db, name, cleanup } = createIntegrationDatabase()
	cleanups.push(cleanup)
	return { db, name }
}

afterEach(cleanups.run)

describe('createDatabase over createIndexedDBDriver', () => {
	it('runs typed CRUD and batch through the core Table', async () => {
		const { db } = database()
		const users = db.table('users')
		const [ada, ...rest] = userRows()
		// Split single-set + batch-set deliberately, to exercise both overloads.
		await users.set(ada)
		await users.set(rest)
		expect(await users.get('u1')).toEqual(createUserRow())
		expect(await users.count()).toBe(3)
		expect((await users.get(['u2', 'u3'])).map((row) => row?.name)).toEqual(['Grace', 'Edsger'])
	})

	it('queries through the core engine over the driver scan', async () => {
		const { db } = database()
		await db.table('users').set(userRows())
		const senior = await db.table('users').query().where('age').from(45).descending('age').all()
		expect(senior.map((row) => row.id)).toEqual(['u3', 'u2'])
	})

	it('persists across close and reopen over the same name', async () => {
		const { db, name } = database()
		await db.table('users').set(createUserRow())
		await db.close()
		const reopened = createDatabase({
			driver: createIndexedDBDriver(name),
			tables: INTEGRATION_TABLES,
		})
		cleanups.push(async () => {
			await reopened.close()
			await deleteDatabase(name)
		})
		expect(await reopened.table('users').get('u1')).toEqual(createUserRow())
	})

	it('rolls a transaction back on throw', async () => {
		const { db } = database()
		const users = db.table('users')
		await users.set(createUserRow())
		const caught = await db
			.transaction(async () => {
				await users.set({ id: 'u1', name: 'Changed', age: 99 })
				await users.set({ id: 'u2', name: 'Ghost', age: 1 })
				throw new Error('boom')
			})
			.catch((error: unknown) => error)
		expect(caught).toBeInstanceOf(Error)
		expect(await users.get('u1')).toEqual(createUserRow())
		expect(await users.get('u2')).toBeUndefined()
	})

	it('loads relations over the driver', async () => {
		const { db } = database()
		await db.table('users').set(createUserRow())
		await db.table('posts').set([
			{ id: 'p1', author: 'u1', title: 'First' },
			{ id: 'p2', author: 'u1', title: 'Second' },
		])
		const manager = createRelationManager({
			database: db,
			relations: INTEGRATION_RELATIONS,
		})
		const ada = await manager.model('users').load('u1', { posts: true })
		expect(ada?.name).toBe('Ada')
		expect(ada?.posts).toHaveLength(2)
	})
})
