import type { TableSchema } from '@src/core'
import { createSQLiteDriver } from '@src/server'
import { afterEach, describe, expect, it } from 'vitest'

// The SQLite driver factory — that `createSQLiteDriver` returns a working
// `DriverInterface`. The full nine-primitive behavior lives in SQLiteDriver.test.ts
// and the whole-stack integration in integration.test.ts; here we only assert the
// factory wires up a usable driver and defaults its path to an in-memory database.

const SCHEMA: readonly TableSchema[] = [
	{
		name: 'users',
		primary: 'id',
		columns: [
			{ name: 'id', type: 'text', nullable: false },
			{ name: 'name', type: 'text', nullable: false },
		],
		indexes: [],
	},
]

let driver = createSQLiteDriver()

afterEach(async () => {
	await driver.close()
})

describe('createSQLiteDriver', () => {
	it('returns a working DriverInterface over the default in-memory database', async () => {
		// No path argument → defaults to ':memory:'; a round-trip proves it is usable.
		driver = createSQLiteDriver()
		await driver.open(SCHEMA)
		await driver.write('users', 'u1', { id: 'u1', name: 'Ada' })
		expect(await driver.read('users', 'u1')).toEqual({ id: 'u1', name: 'Ada' })
		expect(await driver.keys('users')).toEqual(['u1'])
	})

	it('returns a working DriverInterface for an explicit :memory: path', async () => {
		driver = createSQLiteDriver(':memory:')
		await driver.open(SCHEMA)
		await driver.write('users', 'u1', { id: 'u1', name: 'Ada' })
		expect(await driver.read('users', 'u1')).toEqual({ id: 'u1', name: 'Ada' })
	})
})
