import { createDatabase, createRelationManager } from '@src/core'
import { createJSONDriver, createSQLiteDriver } from '@src/server'
import { afterEach, describe, expect, it } from 'vitest'
import {
	createUserRow,
	INTEGRATION_RELATIONS,
	INTEGRATION_TABLES,
	userRows,
} from '../../../setup.js'
import { tempDatabasePath } from '../../../setupServer.js'

// The whole core database + relations stack running over the SQLite driver — the
// proof that the driver is a true drop-in for the in-memory one: the Database /
// Table / Query / relations API is unchanged, only persistence differs. Mirrors
// the browser IndexedDB integration test, sharing its `INTEGRATION_TABLES` /
// `INTEGRATION_RELATIONS` fixtures (tests/setup.ts). Relations need ZERO
// driver-specific code (they flow through Database / Table).

const cleanups: Array<() => void> = []

afterEach(() => {
	for (const clean of cleanups.splice(0)) clean()
})

// An in-memory database for the non-persistence cases.
function database() {
	return createDatabase({ driver: createSQLiteDriver(), tables: INTEGRATION_TABLES })
}

describe('createDatabase over createSQLiteDriver', () => {
	it('runs typed CRUD and batch through the core Table', async () => {
		const users = database().table('users')
		const [ada, ...rest] = userRows()
		// Split single-set + batch-set deliberately, to exercise both overloads.
		await users.set(ada)
		await users.set(rest)
		expect(await users.get('u1')).toEqual(createUserRow())
		expect(await users.count()).toBe(3)
		expect((await users.get(['u2', 'u3'])).map((row) => row?.name)).toEqual(['Grace', 'Edsger'])
	})

	it('queries through the driver natively', async () => {
		const users = database().table('users')
		await users.set(userRows())
		const senior = await users.query().where('age').from(45).descending('age').all()
		expect(senior.map((row) => row.id)).toEqual(['u3', 'u2'])
	})

	it('rolls a transaction back on throw', async () => {
		const db = database()
		const users = db.table('users')
		await users.set(createUserRow())
		const caught = await db
			.transaction(async () => {
				await users.set({ id: 'u1', name: 'Changed', age: 99 })
				await users.set({ id: 'u2', name: 'Ghost', age: 1 })
				throw new Error('boom')
			})
			.catch((error: unknown) => error)
		expect(caught).toBeInstanceOf(Error)
		expect(await users.get('u1')).toEqual(createUserRow())
		expect(await users.get('u2')).toBeUndefined()
	})

	it('persists across close and reopen over a file database', async () => {
		const { path, cleanup } = tempDatabasePath()
		const db = createDatabase({ driver: createSQLiteDriver(path), tables: INTEGRATION_TABLES })
		await db.table('users').set(createUserRow())
		await db.close()
		const reopened = createDatabase({
			driver: createSQLiteDriver(path),
			tables: INTEGRATION_TABLES,
		})
		expect(await reopened.table('users').get('u1')).toEqual(createUserRow())
		// Close the connection BEFORE removing the file (Windows holds an open
		// handle), so the directory cleanup never hits EPERM.
		await reopened.close()
		cleanups.push(cleanup)
	})

	it('loads relations over the driver', async () => {
		const db = database()
		await db.table('users').set(createUserRow())
		await db.table('posts').set([
			{ id: 'p1', author: 'u1', title: 'First' },
			{ id: 'p2', author: 'u1', title: 'Second' },
		])
		const manager = createRelationManager({ database: db, relations: INTEGRATION_RELATIONS })
		const ada = await manager.model('users').load('u1', { posts: true })
		expect(ada?.name).toBe('Ada')
		expect(ada?.posts).toHaveLength(2)
	})
})

describe('createDatabase over createJSONDriver', () => {
	it('persists the whole stack across close and reopen over a JSON file', async () => {
		const { path, cleanup } = tempDatabasePath()
		const db = createDatabase({ driver: createJSONDriver(path), tables: INTEGRATION_TABLES })
		await db
			.table('users')
			.set([createUserRow(), createUserRow({ id: 'u2', name: 'Grace', age: 45 })])
		await db.close()
		const reopened = createDatabase({
			driver: createJSONDriver(path),
			tables: INTEGRATION_TABLES,
		})
		expect(await reopened.table('users').get('u1')).toEqual(createUserRow())
		// Querying runs through the core engine over scan (the JSON driver is scan-only).
		const senior = await reopened.table('users').query().where('age').from(40).all()
		expect(senior.map((row) => row.id)).toEqual(['u2'])
		await reopened.close()
		cleanups.push(cleanup)
	})
})
