import { beforeEach, describe, expect, it } from 'vitest'
import {
	PARITY_ROWS,
	type ParityAggregator,
	type ParityComparator,
	setupParity,
} from '../../../setupServer.js'

// THE SAFETY NET. Over identical data, SQLiteDriver's native records / count
// (Criteria → SQL) must return results IDENTICAL to the core engine path
// (MemoryDriver, which has no native hooks and falls back to applyCriteria over
// scan). A battery of queries — every WHERE operator, and/or mixes, ordering, and
// paging — is run against both and asserted equal, proving native ↔ engine parity.
// `setupParity` (tests/setupServer.ts) stands up both seeded backends and the
// `parity` comparator; the json `payload` column it seeds proves json_extract on
// the native backend agrees with resolveField on the engine for every
// operand-bearing operator (including the present-but-null `note` case).

let parity: ParityComparator
let aggregate: ParityAggregator

beforeEach(async () => {
	;({ parity, aggregate } = await setupParity())
})

describe('SQLiteDriver native records/count parity — operators', () => {
	it('equals', async () => {
		const { native, engine } = await parity((t) => t.query().where('age').equals(36))
		expect(native).toEqual(engine)
	})

	it('not', async () => {
		const { native, engine } = await parity((t) => t.query().where('age').not(36))
		expect(native).toEqual(engine)
	})

	it('above', async () => {
		const { native, engine } = await parity((t) => t.query().where('age').above(40))
		expect(native).toEqual(engine)
	})

	it('below', async () => {
		const { native, engine } = await parity((t) => t.query().where('age').below(40))
		expect(native).toEqual(engine)
	})

	it('from', async () => {
		const { native, engine } = await parity((t) => t.query().where('age').from(41))
		expect(native).toEqual(engine)
	})

	it('to', async () => {
		const { native, engine } = await parity((t) => t.query().where('age').to(36))
		expect(native).toEqual(engine)
	})

	it('between', async () => {
		const { native, engine } = await parity((t) => t.query().where('age').between(30, 45))
		expect(native).toEqual(engine)
	})

	it('like', async () => {
		const { native, engine } = await parity((t) => t.query().where('name').like('A%'))
		expect(native).toEqual(engine)
	})

	it('glob', async () => {
		const { native, engine } = await parity((t) => t.query().where('name').glob('A*'))
		expect(native).toEqual(engine)
	})

	it('starts', async () => {
		const { native, engine } = await parity((t) => t.query().where('name').starts('A'))
		expect(native).toEqual(engine)
	})

	it('ends', async () => {
		const { native, engine } = await parity((t) => t.query().where('name').ends('a'))
		expect(native).toEqual(engine)
	})

	it('any', async () => {
		const { native, engine } = await parity((t) => t.query().where('id').any(['u1', 'u3', 'u5']))
		expect(native).toEqual(engine)
	})

	it('any (empty list matches nothing)', async () => {
		const { native, engine } = await parity((t) => t.query().where('id').any([]))
		expect(native).toEqual(engine)
		expect(native.count).toBe(0)
	})

	it('none', async () => {
		const { native, engine } = await parity((t) => t.query().where('id').none(['u1', 'u3']))
		expect(native).toEqual(engine)
	})

	it('none (empty list matches all)', async () => {
		const { native, engine } = await parity((t) => t.query().where('id').none([]))
		expect(native).toEqual(engine)
		expect(native.count).toBe(PARITY_ROWS.length)
	})

	it('equals on a boolean column', async () => {
		const { native, engine } = await parity((t) => t.query().where('active').equals(true))
		expect(native).toEqual(engine)
	})

	it('absent / present on always-set columns', async () => {
		const absent = await parity((t) => t.query().where('name').absent())
		expect(absent.native).toEqual(absent.engine)
		const present = await parity((t) => t.query().where('name').present())
		expect(present.native).toEqual(present.engine)
	})
})

describe('SQLiteDriver native records/count parity — nested FieldPath (json_extract)', () => {
	// json_extract returns the native scalar (a string unquoted, a number as-is, a
	// JSON boolean as 1/0), so a nested operand must encode to that same scalar —
	// NOT JSON.stringify'd. Each operand-bearing operator is proven against the
	// engine here; these fail before the operand-encoding fix (zero rows on SQLite).

	it('equals on a nested string field', async () => {
		const { native, engine } = await parity((t) =>
			t.query().where(['payload', 'tag']).equals('green'),
		)
		expect(native).toEqual(engine)
		expect(native.count).toBe(1)
	})

	it('not on a nested string field', async () => {
		const { native, engine } = await parity((t) => t.query().where(['payload', 'tag']).not('red%y'))
		expect(native).toEqual(engine)
		expect(native.count).toBe(4)
	})

	it('from on a nested numeric field', async () => {
		const { native, engine } = await parity((t) => t.query().where(['payload', 'score']).from(30))
		expect(native).toEqual(engine)
		expect(native.count).toBe(3)
	})

	it('between on a nested numeric field', async () => {
		const { native, engine } = await parity((t) =>
			t.query().where(['payload', 'score']).between(20, 40),
		)
		expect(native).toEqual(engine)
		expect(native.count).toBe(4)
	})

	it('any on a nested string field', async () => {
		const { native, engine } = await parity((t) =>
			t.query().where(['payload', 'tag']).any(['green', 'blue_z']),
		)
		expect(native).toEqual(engine)
		expect(native.count).toBe(2)
	})

	it('none on a nested string field', async () => {
		const { native, engine } = await parity((t) =>
			t.query().where(['payload', 'tag']).none(['green', 'blue_z']),
		)
		expect(native).toEqual(engine)
		expect(native.count).toBe(4)
	})

	it('equals on a nested boolean field (proves the 1/0 mapping)', async () => {
		// A JSON boolean comes back from json_extract as integer 1/0; the operand
		// `true` must encode to `1` to match, not to JSON.stringify(true) = 'true'.
		const { native, engine } = await parity((t) =>
			t.query().where(['payload', 'flag']).equals(true),
		)
		expect(native).toEqual(engine)
		expect(native.count).toBe(3)
	})

	it('starts on a nested json field whose operand contains a wildcard', async () => {
		// Exercises json_extract cross-backend AND the escapeLike path: 'red_' must
		// match the literal "red_x" (not the LIKE pattern), so SQLite's escaped
		// `LIKE … ESCAPE '\'` over json_extract has to agree with the engine's
		// literal startsWith. Counts: only "red_x" qualifies.
		const starts = await parity((t) => t.query().where(['payload', 'tag']).starts('red_'))
		expect(starts.native).toEqual(starts.engine)
		expect(starts.native.count).toBe(1)
	})

	it('ends on a nested json field whose operand contains a wildcard', async () => {
		// '%y' is a literal suffix here — only the two "red%y" rows end with it, so
		// the escaped LIKE over json_extract must agree with the engine's endsWith.
		const ends = await parity((t) => t.query().where(['payload', 'tag']).ends('%y'))
		expect(ends.native).toEqual(ends.engine)
		expect(ends.native.count).toBe(2)
	})

	it('count of a nested json field query', async () => {
		// A standalone .count() over a json_extract WHERE, confirming the native
		// COUNT path and the engine agree on the same nested-field predicate.
		const ends = await parity((t) => t.query().where(['payload', 'tag']).ends('z'))
		expect(ends.native).toEqual(ends.engine)
		expect(ends.native.count).toBe(1)
	})

	it('equals(null) on a present-but-null nested field', async () => {
		// json_extract collapses a stored JSON `null` to SQL `NULL`, so `= ?` binding
		// NULL never matches under three-valued logic; the fix compiles `IS NULL`,
		// matching the engine, which treats a present-but-null nested value as null.
		// The three `note: null` rows (u2, u4, u6) qualify on both backends.
		const { native, engine } = await parity((t) =>
			t.query().where(['payload', 'note']).equals(null),
		)
		expect(native).toEqual(engine)
		expect(native.count).toBe(3)
	})

	it('not(null) on a present-but-null nested field', async () => {
		// The complement: `IS NOT NULL` matches the three string-valued `note` rows
		// (u1, u3, u5), agreeing with the engine's `not(null)`.
		const { native, engine } = await parity((t) => t.query().where(['payload', 'note']).not(null))
		expect(native).toEqual(engine)
		expect(native.count).toBe(3)
	})
})

describe('SQLiteDriver native records/count parity — and/or mixes', () => {
	it('two ANDs', async () => {
		const { native, engine } = await parity((t) =>
			t.query().where('age').from(30).and('active').equals(true),
		)
		expect(native).toEqual(engine)
	})

	it('two ORs', async () => {
		const { native, engine } = await parity((t) =>
			t.query().where('age').below(25).or('age').above(48),
		)
		expect(native).toEqual(engine)
	})

	it('AND then OR (left-to-right grouping)', async () => {
		const { native, engine } = await parity((t) =>
			t.query().where('age').from(36).and('active').equals(true).or('name').starts('B'),
		)
		expect(native).toEqual(engine)
	})

	it('OR then AND (left-to-right grouping)', async () => {
		const { native, engine } = await parity((t) =>
			t.query().where('name').starts('A').or('age').above(48).and('active').equals(false),
		)
		expect(native).toEqual(engine)
	})
})

describe('SQLiteDriver native records/count parity — ordering and paging', () => {
	it('an UNORDERED query returns rows in KEY order on both backends', async () => {
		// No .ascending/.descending: the contract says scan/records yield in key order.
		// PARITY_ROWS is seeded out of key order (u3, u1, u6, …), so this genuinely
		// proves SQLite (ORDER BY <primary>) and the engine (MemoryDriver, now key-
		// ordered) agree — it would FAIL if MemoryDriver reverted to insertion order.
		const { native, engine } = await parity((t) => t.query())
		expect(native).toEqual(engine)
		expect(native.rows.map((row) => row.id)).toEqual(['u1', 'u2', 'u3', 'u4', 'u5', 'u6'])
	})

	it('order ascending (resolves the age=36 tie in KEY order, not insertion order)', async () => {
		// PARITY_ROWS seeds the age=36 tie KEY-DESCENDING (u6 before u1), so insertion
		// order disagrees with key order on the tie. `.ascending('age')` must therefore
		// place u1 before u6 on BOTH backends: the engine's stable sort over a key-
		// ordered scan, and SQLite's `ORDER BY "age" ASC, "id"` tie-breaker. WITHOUT the
		// compiled primary tie-breaker SQLite would order the tie by rowid (insertion =
		// u6, u1) and DIVERGE — so this case fails on a revert, proving the fix genuine.
		const { native, engine } = await parity((t) => t.query().ascending('age'))
		expect(native).toEqual(engine)
		expect(native.rows.map((row) => row.id)).toEqual(['u5', 'u1', 'u6', 'u2', 'u4', 'u3'])
	})

	it('order descending then ascending (tie-break)', async () => {
		const { native, engine } = await parity((t) => t.query().descending('age').ascending('name'))
		expect(native).toEqual(engine)
	})

	it('limit', async () => {
		const { native, engine } = await parity((t) => t.query().ascending('age').limit(3))
		expect(native).toEqual(engine)
	})

	it('offset', async () => {
		const { native, engine } = await parity((t) => t.query().ascending('age').offset(2))
		expect(native).toEqual(engine)
	})

	it('limit and offset', async () => {
		const { native, engine } = await parity((t) => t.query().ascending('age').limit(2).offset(1))
		expect(native).toEqual(engine)
	})

	it('filtered, ordered, and paged together', async () => {
		const { native, engine } = await parity((t) =>
			t.query().where('active').equals(true).descending('age').limit(2),
		)
		expect(native).toEqual(engine)
	})
})

describe('SQLiteDriver native aggregate parity — vs the core engine', () => {
	// The aggregate safety net: SQLiteDriver's SQL COUNT/SUM/AVG/MIN/MAX (the native
	// `aggregate?` hook) must deep-equal the core engine's `computeAggregate` over the
	// MemoryDriver (no hook → scan + reduce) for the same numeric column and the same
	// rows. `age` is a clean integer column, matching how the operator cases use typed
	// columns — a mixed/non-numeric column is a documented divergence, not tested here.

	it('count over the whole table equals the engine (COUNT(*) = all rows)', async () => {
		// Proves COUNT(*), not COUNT(age): every row is counted, matching the engine's
		// rows.length, even though COUNT(col) would only differ on a null cell.
		const { native, engine } = await aggregate('count', 'age')
		expect(native).toEqual(engine)
		expect(native).toBe(PARITY_ROWS.length)
	})

	it('sum over the whole table equals the engine', async () => {
		const { native, engine } = await aggregate('sum', 'age')
		expect(native).toEqual(engine)
	})

	it('average over the whole table equals the engine', async () => {
		const { native, engine } = await aggregate('average', 'age')
		expect(native).toEqual(engine)
	})

	it('minimum over the whole table equals the engine', async () => {
		const { native, engine } = await aggregate('minimum', 'age')
		expect(native).toEqual(engine)
	})

	it('maximum over the whole table equals the engine', async () => {
		const { native, engine } = await aggregate('maximum', 'age')
		expect(native).toEqual(engine)
	})

	it('every aggregate under a WHERE condition equals the engine', async () => {
		// The three active rows constrain each aggregate; native SQL over the compiled
		// WHERE must still match the engine's filter-then-reduce.
		const scope: Parameters<ParityAggregator>[2] = (t) => t.query().where('active').equals(true)
		for (const operation of ['count', 'sum', 'average', 'minimum', 'maximum'] as const) {
			const { native, engine } = await aggregate(operation, 'age', scope)
			expect(native).toEqual(engine)
		}
	})

	it('every aggregate over zero matched rows equals the engine (0 / undefined)', async () => {
		// No row has age > 100: count must be 0 on both, and SUM/AVG/MIN/MAX must be
		// undefined on both (SQL NULL → undefined, matching the engine's empty result).
		const scope: Parameters<ParityAggregator>[2] = (t) => t.query().where('age').above(100)
		const counted = await aggregate('count', 'age', scope)
		expect(counted.native).toEqual(counted.engine)
		expect(counted.native).toBe(0)
		for (const operation of ['sum', 'average', 'minimum', 'maximum'] as const) {
			const { native, engine } = await aggregate(operation, 'age', scope)
			expect(native).toEqual(engine)
			expect(native).toBeUndefined()
		}
	})
})
